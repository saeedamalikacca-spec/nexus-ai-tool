require("dotenv").config();

const express = require("express");
const OpenAI = require("openai");
const path = require("path");

const app = express();
const PORT = 3000;

app.use(express.json({ limit: "20kb" }));
app.use(express.static(__dirname));

const client = new OpenAI({
  apiKey: process.env.OPENAI_API_KEY,
});

app.post("/api/ask", async (req, res) => {
  try {
    const { message } = req.body;

    if (typeof message !== "string" || !message.trim()) {
      return res.status(400).json({
        error: "Please enter a message.",
      });
    }

    if (message.length > 4000) {
      return res.status(400).json({
        error: "Your message is too long.",
      });
    }

    const response = await client.responses.create({
      model: "gpt-4.1-mini",
      instructions:
        "You are NEXUS AI, a helpful assistant. Give clear, concise answers. When useful, structure your answer with headings or bullet points.",
      input: message,
    });

    res.json({
      answer: response.output_text,
    });
  } catch (error) {
    console.error("OpenAI request failed:", error.status || "unknown status");

    res.status(500).json({
      error: "The AI request failed. Check the server configuration and API access.",
    });
  }
});

app.listen(PORT, () => {
  console.log(`NEXUS AI is running at http://localhost:${PORT}`);
});
